"use client";

let accessToken: string | null = null;

export function setAccessToken(token: string | null) {
  accessToken = token;
}

export function getAccessToken() {
  return accessToken;
}

async function tryRefresh(): Promise<string | null> {
  try {
    const res = await fetch("/api/auth/refresh", {
      method: "POST",
      credentials: "include",
    });
    if (!res.ok) return null;
    const data = (await res.json()) as { accessToken?: string };
    if (data.accessToken) {
      accessToken = data.accessToken;
      return data.accessToken;
    }
    return null;
  } catch {
    return null;
  }
}

export interface ApiOptions extends RequestInit {
  retryOn401?: boolean;
  extraHeaders?: Record<string, string>;
}

export async function apiFetch<T>(path: string, opts: ApiOptions = {}): Promise<T> {
  const { retryOn401 = true, extraHeaders, ...init } = opts;

  const doFetch = (token: string | null) => {
    const headers = new Headers(init.headers || {});
    if (token) headers.set("Authorization", `Bearer ${token}`);
    if (init.body && !(init.body instanceof FormData) && !headers.has("Content-Type")) {
      headers.set("Content-Type", "application/json");
    }
    if (extraHeaders) {
      for (const [k, v] of Object.entries(extraHeaders)) headers.set(k, v);
    }
    return fetch(path, { ...init, headers, credentials: "include" });
  };

  let res = await doFetch(accessToken);
  if (res.status === 401 && retryOn401) {
    const refreshed = await tryRefresh();
    if (refreshed) res = await doFetch(refreshed);
  }

  if (!res.ok) {
    const text = await res.text().catch(() => "");
    throw Object.assign(new Error(text || `Request failed (${res.status})`), {
      status: res.status,
    });
  }
  if (res.status === 204) return undefined as T;
  return (await res.json()) as T;
}

/* Convenience wrappers */
export const api = {
  get: <T>(p: string, opts?: ApiOptions) => apiFetch<T>(p, { ...opts, method: "GET" }),
  post: <T>(p: string, body?: unknown, opts?: ApiOptions) =>
    apiFetch<T>(p, {
      ...opts,
      method: "POST",
      body: body instanceof FormData ? body : body ? JSON.stringify(body) : undefined,
    }),
  patch: <T>(p: string, body?: unknown, opts?: ApiOptions) =>
    apiFetch<T>(p, {
      ...opts,
      method: "PATCH",
      body: body instanceof FormData ? body : body ? JSON.stringify(body) : undefined,
    }),
  del: <T>(p: string, opts?: ApiOptions) => apiFetch<T>(p, { ...opts, method: "DELETE" }),
};

/* SSE / polling helper for run progress */
export function subscribeRunProgress(
  runId: string,
  onEvent: (data: { stage: string; status: Run["status"] }) => void
): () => void {
  // Prefer SSE if available; fall back to polling every 2s.
  let closed = false;
  let es: EventSource | null = null;
  let interval: ReturnType<typeof setInterval> | null = null;

  const startPolling = () => {
    interval = setInterval(async () => {
      if (closed) return;
      try {
        const r = await api.get<{ status: Run["status"]; stageLabel?: string }>(
          `/api/runs/${runId}`
        );
        onEvent({ stage: r.stageLabel || "", status: r.status });
        if (r.status === "review" || r.status === "completed" || r.status === "failed") {
          cleanup();
        }
      } catch {
        /* noop — keep polling */
      }
    }, 2000);
  };

  try {
    es = new EventSource(`/api/runs/${runId}/stream`);
    es.onmessage = (e) => {
      try {
        const data = JSON.parse(e.data) as { stage: string; status: Run["status"] };
        onEvent(data);
        if (data.status === "review" || data.status === "completed" || data.status === "failed") {
          cleanup();
        }
      } catch {
        /* ignore malformed */
      }
    };
    es.onerror = () => {
      es?.close();
      es = null;
      if (!closed) startPolling();
    };
  } catch {
    startPolling();
  }

  const cleanup = () => {
    closed = true;
    es?.close();
    if (interval) clearInterval(interval);
  };

  return cleanup;
}

"use client";

import { createContext, useContext, useEffect, useState, useCallback } from "react";
import { api, setAccessToken } from "./api";
import type { User } from "./types";

interface AuthContextValue {
  user: User | null;
  loading: boolean;
  login: (email: string, password: string) => Promise<void>;
  logout: () => Promise<void>;
  refresh: () => Promise<void>;
}

const AuthContext = createContext<AuthContextValue | null>(null);

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);

  const refresh = useCallback(async () => {
    try {
      const res = await api.post<{ accessToken: string; user: User }>("/api/auth/refresh");
      setAccessToken(res.accessToken);
      setUser(res.user);
    } catch {
      setAccessToken(null);
      setUser(null);
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    void refresh();
  }, [refresh]);

  const login = useCallback(async (email: string, password: string) => {
    const res = await api.post<{ accessToken: string; user: User }>("/api/auth/login", {
      email,
      password,
    });
    setAccessToken(res.accessToken);
    setUser(res.user);
  }, []);

  const logout = useCallback(async () => {
    try {
      await api.post("/api/auth/logout");
    } finally {
      setAccessToken(null);
      setUser(null);
    }
  }, []);

  return (
    <AuthContext.Provider value={{ user, loading, login, logout, refresh }}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error("useAuth must be used inside <AuthProvider>");
  return ctx;
}

export function Button({
  variant = "primary",
  loading,
  className,
  children,
  ...props
}: ButtonHTMLAttributes<HTMLButtonElement> & {
  variant?: "primary" | "ghost" | "outline" | "danger";
  loading?: boolean;
}) {
  const map = {
    primary: "btn-primary",
    ghost:   "btn-ghost",
    outline: "btn-outline",
    danger:  "btn-danger",
  };
  return (
    <button
      {...props}
      disabled={props.disabled || loading}
      className={clsx("btn", map[variant], className, "disabled:cursor-not-allowed disabled:opacity-50")}
    >
      {loading && <Loader2 size={14} className="animate-spin" />}
      {children}
    </button>
  );
}

import { useEffect } from "react";

export default function AppLayout({ children }: { children: React.ReactNode }) {
  const { user, loading } = useAuth();
  const router = useRouter();

  useEffect(() => {
    if (!loading && !user) router.replace("/login");
  }, [loading, user, router]);

  if (loading) {
    return (
      <div className="flex min-h-screen items-center justify-center text-fg-2">Loading…</div>
    );
  }
  if (!user) return null;

  return <AppShell>{children}</AppShell>;
}

import { useEffect, useState } from "react";

import { useParams, useRouter } from "next/navigation";
import { api, subscribeRunProgress } from "@/lib/api";
import { Badge, Card, Spinner } from "@/components/ui";
import type { Run } from "@/lib/types";

const STAGES = [
  "Ingest JD",
  "Parse profile",
  "Draft sections",
  "Validate",
  "Render",
];

export default function RunProgressPage() {
  const { id } = useParams<{ id: string }>();
  const router = useRouter();

  const { data: run } = useQuery({
    queryKey: ["run", id],
    queryFn: () => api.get<Run>(`/api/runs/${id}`),
    refetchOnMount: "always",
  });

  const [status, setStatus] = useState<Run["status"]>("queued");
  const [stage, setStage] = useState("");

  useEffect(() => {
    if (!id) return;
    const stop = subscribeRunProgress(id, (evt) => {
      setStatus(evt.status);
      setStage(evt.stage);
      if (evt.status === "review") router.replace(`/runs/${id}/review`);
      if (evt.status === "failed") router.replace(`/runs/${id}`);
    });
    return () => stop();
  }, [id, router]);

  const activeIdx = Math.max(0, STAGES.findIndex((s) => stage.toLowerCase().includes(s.toLowerCase())));

  return (
    <div className="space-y-8">
      <header className="flex items-center justify-between">
        <div>
          <h1 className="text-2xl font-semibold tracking-tight">Run {id.slice(0, 8)}</h1>
          <p className="mt-1 text-sm text-fg-2">
            {run?.profileName ? `Profile: ${run.profileName}` : "Generating your résumé…"}
          </p>
        </div>
        <Badge tone={status === "failed" ? "fail" : status === "review" ? "ok" : "idle"}>
          {status}
        </Badge>
      </header>

      <Card className="space-y-4 p-6">
        {STAGES.map((label, i) => {
          const done = i < activeIdx || status === "review";
          const active = i === activeIdx && status !== "review";
          return (
            <div key={label} className="flex items-center gap-3 text-sm">
              <div
                className={
                  "flex h-6 w-6 items-center justify-center rounded-full border text-[11px] " +
                  (done
                    ? "border-success/40 bg-success/10 text-success"
                    : active
                    ? "border-gold/40 bg-gold/10 text-gold-2"
                    : "border-border bg-bg-3 text-fg-2")
                }
              >
                {done ? "✓" : i + 1}
              </div>
              <span className={done ? "text-fg" : active ? "text-gold-2" : "text-fg-2"}>
                {label}
              </span>
              {active && <Spinner className="ml-auto" />}
            </div>
          );
        })}
      </Card>
    </div>
  );
}

function SectionEditor({
  runId, section, onMutate,
}: { runId: string; section: Section; onMutate: () => void }) {

  );
}

function BulletRow({ runId, bullet, onMutate }: { runId: string; bullet: Bullet; onMutate: () => void }) {
  const [editing, setEditing] = useState(false);
  const [text, setText] = useState(bullet.text);

  const save = useMutation({
    mutationFn: () => api.patch(`/api/runs/${runId}/bullets/${bullet.id}`, { text }),
    onSuccess: () => { setEditing(false); onMutate(); },
  });

  return (
    <li className="group rounded-lg border border-border bg-bg-2 p-3 text-[13px] leading-relaxed text-fg">
      {editing ? (
        <div className="space-y-2">
          <textarea

            <Button onClick={() => save.mutate()} loading={save.isPending}>Save</Button>

              Cancel
            </Button>
          </div>
        </div>
      ) : (
        <div className="flex gap-2">
          <span className="flex-1">{bullet.text}</span>
          <button
            onClick={() => setEditing(true)}
            className="opacity-0 transition-opacity group-hover:opacity-100"
            aria-label="Edit bullet"
          >
            <Pencil size={12} className="text-fg-2" />
          </button>
        </div>
      )}
    </li>
  );
}

function FlagRow({ flag }: { flag: ValidationFlag }) {

      <Badge tone={tone as "fail" | "warn" | "idle"}>{flag.rule}</Badge>

    </li>
  );
}

"use client";

import { useAuth } from "@/lib/auth";
import { useRouter } from "next/navigation";
import { useState } from "react";
import { Button, Card, Input } from "@/components/ui";

export default function LoginPage() {
  const { login } = useAuth();
  const router = useRouter();
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);

  async function onSubmit(e: React.FormEvent) {
    e.preventDefault();
    setError(null);
    setLoading(true);
    try {
      await login(email, password);
      router.replace("/dashboard");
    } catch (err) {
      setError((err as Error).message || "Login failed");
    } finally {
      setLoading(false);
    }
  }

  return (
    <div className="flex min-h-screen items-center justify-center p-6">
      <Card className="w-full max-w-sm space-y-5 p-7">
        <div className="flex items-center gap-2">
          <div className="h-7 w-7 rounded-md bg-gradient-to-br from-gold to-gold-2" />
          <span className="font-semibold">Resume Engine</span>
        </div>

        <form onSubmit={onSubmit} className="space-y-3">
          <Input
            type="email"
            placeholder="you@agency.com"
            value={email}
            onChange={(e) => setEmail(e.target.value)}
            required
          />
          <Input
            type="password"
            placeholder="Password"
            value={password}
            onChange={(e) => setPassword(e.target.value)}
            required
          />
          {error && <p className="text-xs text-danger">{error}</p>}
          <Button type="submit" loading={loading} className="w-full">Sign in</Button>
        </form>
      </Card>
    </div>
  );
}
