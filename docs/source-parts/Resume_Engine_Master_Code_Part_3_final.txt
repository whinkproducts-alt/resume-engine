// packages/rules/src/util/date.ts
// Lenient month-precision date parsing shared by R4, R5, R6, R17.

export interface YearMonth {
  y: number;
  m: number; // 1..12
}

const ISO_MONTH = /^(\d{4})-(\d{2})$/;
const ISO_DAY = /^(\d{4})-(\d{2})-(\d{2})$/;
const YEAR_ONLY = /^(\d{4})$/;
const PRESENT = /^(present|current|now|ongoing)$/i;

/**
 * Parses 'YYYY-MM', 'YYYY-MM-DD', 'YYYY', 'Present', 'Current'.
 * Returns null when the value cannot be interpreted.
 */
export function parseYearMonth(raw: string | null | undefined): YearMonth | null {
  if (raw == null) return null;
  const s = String(raw).trim();
  if (!s) return null;
  if (PRESENT.test(s)) return null; // caller treats as "open ended"

  let m = ISO_MONTH.exec(s);
  if (m) return normalise(Number(m[1]), Number(m[2]));

  m = ISO_DAY.exec(s);
  if (m) return normalise(Number(m[1]), Number(m[2]));

  m = YEAR_ONLY.exec(s);
  if (m) return normalise(Number(m[1]), 1);

  // 'Jan 2021' / 'January 2021' / '2021-1'
  const loose = /^([A-Za-z]{3,9})[\s,/-]+(\d{4})$/.exec(s);
  if (loose) {
    const monthIdx = MONTH_NAMES.indexOf(loose[1].slice(0, 3).toLowerCase());
    if (monthIdx >= 0) return normalise(Number(loose[2]), monthIdx + 1);
  }
  const reversed = /^(\d{4})[\s,/-]+([A-Za-z]{3,9})$/.exec(s);
  if (reversed) {
    const monthIdx = MONTH_NAMES.indexOf(reversed[2].slice(0, 3).toLowerCase());
    if (monthIdx >= 0) return normalise(Number(reversed[1]), monthIdx + 1);
  }
  const shortMonth = /^(\d{4})-(\d{1,2})$/.exec(s);
  if (shortMonth) return normalise(Number(shortMonth[1]), Number(shortMonth[2]));

  return null;
}

const MONTH_NAMES = [
  'jan', 'feb', 'mar', 'apr', 'may', 'jun',
  'jul', 'aug', 'sep', 'oct', 'nov', 'dec',
];

function normalise(y: number, m: number): YearMonth | null {
  if (!Number.isFinite(y) || !Number.isFinite(m)) return null;
  if (y < 1950 || y > 2100) return null;
  if (m < 1 || m > 12) return null;
  return { y, m };
}

export function toMonths(vm: YearMonth): number {
  return vm.y * 12 + (vm.m - 1);
}

export function monthsBetween(a: YearMonth, b: YearMonth): number {
  return toMonths(b) - toMonths(a);
}

export function isPresent(raw: string | null | undefined): boolean {
  return raw != null && PRESENT.test(String(raw).trim());
}

export function isCanonicalIsoMonth(raw: string | null | undefined): boolean {
  return raw != null && ISO_MONTH.test(String(raw).trim());
}

export function formatYearMonth(vm: YearMonth): string {
  const mm = String(vm.m).padStart(2, '0');
  return `${vm.y}-${mm}`;
}

// packages/rules/src/util/text.ts
// Tokenising, verb extraction, and numeric-token scanning.

/* ------------------------------ R2 tokenising ------------------------------ */

export function tokenize(input: string): string[] {
  return String(input ?? '')
    .toLowerCase()
    .replace(/[^a-z0-9\s+#.]/g, ' ')
    .replace(/\./g, ' ')
    .split(/\s+/)
    .filter(Boolean);
}

/**
 * Overlap coefficient (intersection / smaller-set-size).
 * Chosen over Jaccard so that "Data Scientist" vs "Lead Data Scientist"
 * scores 1.0 once stop words are removed, which is the intent of the
 * 70% close-variant rule.
 */
export function tokenOverlap(
  a: string,
  b: string,
  stopWords: readonly string[],
): { score: number; shared: string[]; onlyA: string[]; onlyB: string[] } {
  const stop = new Set(stopWords.map((w) => w.toLowerCase()));
  const setA = new Set(tokenize(a).filter((t) => !stop.has(t)));
  const setB = new Set(tokenize(b).filter((t) => !stop.has(t)));

  if (setA.size === 0 || setB.size === 0) {
    return { score: 0, shared: [], onlyA: [...setA], onlyB: [...setB] };
  }

  const shared = [...setA].filter((t) => setB.has(t));
  const denom = Math.min(setA.size, setB.size);
  return {
    score: shared.length / denom,
    shared,
    onlyA: [...setA].filter((t) => !setB.has(t)),
    onlyB: [...setB].filter((t) => !setA.has(t)),
  };
}

/* ---------------------------- R10 verb extraction --------------------------- */

const IRREGULAR_PAST: Record<string, string> = {
  led: 'lead', built: 'build', drove: 'drive', grew: 'grow',
  ran: 'run', wrote: 'write', made: 'make', began: 'begin',
  brought: 'bring', chose: 'choose', dealt: 'deal', did: 'do',
  found: 'find', gave: 'give', held: 'hold', kept: 'keep',
  knew: 'know', left: 'leave', lost: 'lose', meant: 'mean',
  met: 'meet', paid: 'pay', put: 'put', said: 'say', saw: 'see',
  sold: 'sell', sent: 'send', set: 'set', spent: 'spend',
  took: 'take', taught: 'teach', told: 'tell', thought: 'think',
  won: 'win', wore: 'wear', sped: 'speed', cut: 'cut',
};

/** First token of a bullet, lower-cased, de-punctuated, de-inflected. */
export function extractLeadingVerb(bullet: string): string | null {
  const tokens = tokenize(bullet);
  if (tokens.length === 0) return null;
  const raw = tokens[0];
  if (IRREGULAR_PAST[raw]) return IRREGULAR_PAST[raw];
  if (raw.length > 4 && raw.endsWith('ing')) return raw.slice(0, -3);
  if (raw.length > 3 && raw.endsWith('ed')) return raw.slice(0, -2);
  if (raw.length > 3 && raw.endsWith('s') && !raw.endsWith('ss')) {
    return raw.slice(0, -1);
  }
  return raw;
}

/* --------------------------- R20 numeric scanning --------------------------- */

export interface NumericToken {
  raw: string;
  numeric: number;
  unit: string | null;
  index: number;
}

const NUMERIC_RE =
  /(\d[\d,]*(?:\.\d+)?)\s*(%|percent|k\b|m\b|mm\b|bn\b|b\b|x\b|hrs?\b|hours?\b|days?\b|weeks?\b|months?\b|years?\b|yrs?\b|users?\b|customers?\b|clients?\b|people\b|engineers?\b|developers?\b|reps?\b|requests?\b|rps\b|qps\b|ms\b|tb\b|gb\b|mb\b|usd\b|\$)?/gi;

const UNIT_ALIASES: Record<string, string> = {
  percent: '%',
  hrs: 'hours', hr: 'hours',
  yrs: 'years', yr: 'years',
  rps: 'requests',
  usd: '$',
};

export function scanNumericTokens(text: string): NumericToken[] {
  const out: NumericToken[] = [];
  NUMERIC_RE.lastIndex = 0;
  let m: RegExpExecArray | null;
  while ((m = NUMERIC_RE.exec(text)) !== null) {
    const digits = m[1].replace(/,/g, '');
    const numeric = Number(digits);
    if (!Number.isFinite(numeric)) continue;
    let unit = m[2] ? m[2].toLowerCase().replace(/\b\b/g, '') : null;
    if (unit) {
      unit = unit.replace(/[.,;:]$/, '');
      unit = UNIT_ALIASES[unit] ?? unit;
    }
    out.push({ raw: m[0].trim(), numeric, unit, index: m.index });
    if (m.index === NUMERIC_RE.lastIndex) NUMERIC_RE.lastIndex++; // guard
  }
  return out;
}

/** Expands 'k' / 'm' / 'bn' / '%' suffixes into a canonical magnitude. */
export function canonicalMagnitude(t: NumericToken): number {
  switch (t.unit) {
    case 'k': return t.numeric * 1_000;
    case 'm': case 'mm': return t.numeric * 1_000_000;
    case 'bn': case 'b': return t.numeric * 1_000_000_000;
    default: return t.numeric;
  }
}

export function wordCount(text: string): number {
  return String(text ?? '').trim().split(/\s+/).filter(Boolean).length;
}

export function containsPhrase(haystack: string, phrase: string): boolean {
  const escaped = phrase.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  return new RegExp(`(^|[^a-z0-9])${escaped}([^a-z0-9]|$)`, 'i').test(haystack);
}

// packages/rules/src/fact-bank.ts
// Fact Bank matching used by R20.

import {
  FactBankEntry,
  NumericToken,
} from './types';
import { canonicalMagnitude } from './util/text';

export interface FactMatch {
  token: NumericToken;
  entry: FactBankEntry | null;
  /** How the match was established. */
  via: 'numeric+unit' | 'numeric' | 'alias' | 'raw' | 'none';
  verified: boolean;
}

const UNIT_EQUIVALENTS: Record<string, string> = {
  '%': '%',
  percent: '%',
  k: 'k',
  m: 'm',
  mm: 'mm',
  bn: 'bn',
  b: 'b',
  x: 'x',
  hours: 'hours',
  days: 'days',
  weeks: 'weeks',
  months: 'months',
  years: 'years',
  users: 'users',
  customers: 'customers',
  clients: 'clients',
  people: 'people',
  engineers: 'engineers',
  developers: 'developers',
  reps: 'reps',
  requests: 'requests',
  ms: 'ms',
  tb: 'tb',
  gb: 'gb',
  mb: 'mb',
  '$': '$',
};

function unitKey(u: string | null | undefined): string | null {
  if (!u) return null;
  const k = u.toLowerCase();
  return UNIT_EQUIVALENTS[k] ?? k;
}

/**
 * Finds the Fact Bank entry that best substantiates a numeric token.
 * Precedence: numeric+unit exact -> numeric (unit-agnostic) -> alias -> raw.
 */
export function matchNumericToken(
  token: NumericToken,
  factBank: FactBankEntry[],
): FactMatch {
  const tokenUnit = unitKey(token.unit);
  const magnitude = canonicalMagnitude(token);

  // 1. numeric + unit
  for (const e of factBank) {
    if (e.kind !== 'metric') continue;
    if (e.numeric == null) continue;
    if (Math.abs(e.numeric - magnitude) > 1e-9) continue;
    if (unitKey(e.unit) === tokenUnit) {
      return { token, entry: e, via: 'numeric+unit', verified: e.verified };
    }
  }

  // 2. numeric only (unit-agnostic)
  for (const e of factBank) {
    if (e.kind !== 'metric') continue;
    if (e.numeric == null) continue;
    if (Math.abs(e.numeric - magnitude) <= 1e-9) {
      return { token, entry: e, via: 'numeric', verified: e.verified };
    }
  }

  // 3. alias containment on non-metric facts (e.g. years in date facts)
  const needle = token.raw.toLowerCase();
  for (const e of factBank) {
    const surfaces = [e.value, ...(e.aliases ?? [])].map((s) => s.toLowerCase());
    if (surfaces.some((s) => s === needle || s.includes(needle) || needle.includes(s))) {
      return { token, entry: e, via: 'alias', verified: e.verified };
    }
  }

  // 4. raw value equality against any kind
  for (const e of factBank) {
    if (e.value.trim().toLowerCase() === needle) {
      return { token, entry: e, via: 'raw', verified: e.verified };
    }
  }

  return { token, entry: null, via: 'none', verified: false };
}

function allBullets(ctx: RuleContext): { path: string; text: string }[] {
  const out: { path: string; text: string }[] = [];
  ctx.resume.experience.forEach((exp, i) => {
    (exp.bullets ?? []).forEach((b, j) => {
      out.push({ path: `experience[${i}].bullets[${j}]`, text: b });

    (p.bullets ?? []).forEach((b, j) => {
      out.push({ path: `projects[${i}].bullets[${j}]`, text: b });
    });
  });
  return out;
}

export const R1: Rule = {
  id: 'R1',
  name: 'Required sections present',
  severity: 'error',
  blocking: true,
  evaluate(ctx) {
    const flags: ValidationFlag[] = [];
    const r = ctx.resume;

    if (!r.contact?.name?.trim()) {
      flags.push(flag('R1', 'MISSING_SECTION', 'error', true,
        'Contact name is required.', 'contact.name'));
    }
    if (!r.summary?.trim() || wordCount(r.summary) < 15) {
      flags.push(flag('R1', 'MISSING_SECTION', 'error', true,
        'Professional summary must be present and at least 15 words.',
        'summary'));
    }
    if (!Array.isArray(r.skills) || r.skills.length === 0) {
      flags.push(flag('R1', 'MISSING_SECTION', 'error', true,
        'At least one skills group is required.', 'skills'));
    }
    if (!Array.isArray(r.experience) || r.experience.length === 0) {
      flags.push(flag('R1', 'MISSING_SECTION', 'error', true,
        'At least one experience entry is required.', 'experience'));
    }

export const R2: Rule = {
  id: 'R2',
  name: 'Target role alignment (70% token overlap, stop words ignored)',
  severity: 'error',
  blocking: true,
  evaluate(ctx) {
    const target = ctx.targetRole ?? ctx.resume.meta.targetRole ?? '';
    const headline = ctx.resume.headline ?? '';

    if (!headline.trim()) {
      return [flag('R2', 'ROLE_MISSING', 'error', true,
        'Headline is required for role-alignment checking.', 'headline')];
    }

    const { score, shared, onlyA, onlyB } = tokenOverlap(
      headline, target, ctx.config.roleStopWords,
    );

    const norm = (s: string) => s.trim().toLowerCase().replace(/\s+/g, ' ');
    if (norm(headline) === norm(target)) {
      return []; // exact
    }

    if (score >= ctx.config.roleOverlapThreshold) {
      // Authoritative close-variant behaviour: warn, do not block.
      return [flag('R2', 'ROLE_CLOSE_VARIANT', 'warn', false,
        `Headline "${headline}" is a close variant of target role "${target}" ` +
        `(${(score * 100).toFixed(0)}% token overlap).`,
        'headline',
        { score, shared, onlyA, onlyB, threshold: ctx.config.roleOverlapThreshold })];
    }

    return [flag('R2', 'ROLE_MISMATCH', 'error', true,
      `Headline "${headline}" does not match target role "${target}" ` +
      `(${(score * 100).toFixed(0)}% token overlap, required ` +
      `${(ctx.config.roleOverlapThreshold * 100).toFixed(0)}%).`,
      'headline',
      { score, shared, onlyA, onlyB })];
  },
};

export const R4: Rule = {
  id: 'R4',
  name: 'Chronological consistency',
  severity: 'error',
  blocking: true,
  evaluate(ctx) {
    const flags: ValidationFlag[] = [];
    const nowMonths = (() => {
      const d = new Date();
      return d.getUTCFullYear() * 12 + d.getUTCMonth();
    })();

    ctx.resume.experience.forEach((exp, i) => {
      const start = parseYearMonth(exp.startDate);
      const end = isOpenEnded(exp) ? null : parseYearMonth(exp.endDate ?? null);
      const p = `experience[${i}]`;

      if (!start) {
        flags.push(flag('R4', 'DATE_UNPARSEABLE', 'error', true,
          `Start date "${exp.startDate}" on ${exp.title} @ ${exp.company} ` +
          'cannot be parsed.', `${p}.startDate`));
        return;
      }
      if (!isOpenEnded(exp) && !end) {
        flags.push(flag('R4', 'DATE_UNPARSEABLE', 'error', true,
          `End date "${exp.endDate ?? ''}" on ${exp.title} @ ${exp.company} ` +
          'cannot be parsed (use "Present" for the current role).',
          `${p}.endDate`));
        return;
      }
      if (end && toMonths(end) < toMonths(start)) {
        flags.push(flag('R4', 'DATE_REVERSED', 'error', true,
          `End date precedes start date on ${exp.title} @ ${exp.company}.`,
          `${p}`,
          { startDate: exp.startDate, endDate: exp.endDate }));
      }
      if (end && toMonths(end) > nowMonths) {
        flags.push(flag('R4', 'DATE_FUTURE', 'error', true,
          `End date on ${exp.title} @ ${exp.company} is in the future.`,
          `${p}.endDate`));
      }
    });

    // Overlap detection across roles (sorted by start, ignoring open-ended past).
    const spans = ctx.resume.experience
      .map((exp, i) => {
        const s = parseYearMonth(exp.startDate);
        const e = isOpenEnded(exp) ? null : parseYearMonth(exp.endDate ?? null);
        return s ? { i, s: toMonths(s), e: e ? toMonths(e) : Number.MAX_SAFE_INTEGER,
          label: `${exp.title} @ ${exp.company}` } : null;
      })
      .filter((x): x is NonNullable<typeof x> => x !== null)
      .sort((a, b) => a.s - b.s);

    for (let k = 1; k < spans.length; k++) {
      const prev = spans[k - 1];
      const cur = spans[k];
      if (cur.s < prev.e) {
        flags.push(flag('R4', 'DATE_OVERLAP', 'error', true,
          `Roles overlap: "${prev.label}" and "${cur.label}".`,
          `experience[${cur.i}]`,
          { previous: prev.label, current: cur.label }));
      }
    }

    // Education ordering.
    ctx.resume.education.forEach((edu, i) => {
      const s = parseYearMonth(edu.startDate ?? null);
      const e = parseYearMonth(edu.endDate ?? null);
      if (s && e && toMonths(e) < toMonths(s)) {
        flags.push(flag('R4', 'DATE_REVERSED', 'error', true,
          `End date precedes start date for ${edu.degree} at ${edu.institution}.`,
          `education[${i}]`));
      }
    });

    return flags;
  },
};

export const R6: Rule = {
  id: 'R6',
  name: 'Employment gap detection',
  severity: 'warn',
  blocking: false,
  evaluate(ctx) {
    const flags: ValidationFlag[] = [];
    const spans = ctx.resume.experience
      .map((exp, i) => {
        const s = parseYearMonth(exp.startDate);
        const e = isOpenEnded(exp) ? null : parseYearMonth(exp.endDate ?? null);

          `${gap}-month gap between "${prev.exp.title} @ ${prev.exp.company}" ` +
          `and "${spans[k].exp.title} @ ${spans[k].exp.company}".`,
          `experience[${spans[k].i}]`,
          { gapMonths: gap, threshold: ctx.config.maxGapMonths }));
      }
    }
    return flags;
  },

// packages/rules/src/validator.ts
// Deterministic rule executor. Runs R1..R20, partitions flags, decides retry.

import {
  Rule,
  RuleConfig,
  RuleContext,
  RuleOutcome,
  ValidationFlag,
  ValidationResult,
} from './types';
import { DEFAULT_RULE_CONFIG } from './types';
import { ALL_RULES } from './rules';

export class RuleValidator {
  private readonly rules: Rule[];

  constructor(
    private readonly config: RuleConfig = DEFAULT_RULE_CONFIG,
    rules: Rule[] = ALL_RULES,
  ) {
    this.rules = [...rules];
  }

  register(rule: Rule): this {
    this.rules.push(rule);
    return this;
  }

  /** Replaces a rule by id — used for tenant-specific overrides. */
  override(rule: Rule): this {
    const idx = this.rules.findIndex((r) => r.id === rule.id);
    if (idx >= 0) this.rules[idx] = rule;
    else this.rules.push(rule);
    return this;
  }

  async validate(ctx: Omit<RuleContext, 'config'>): Promise<ValidationResult> {
    const fullCtx: RuleContext = {
      ...ctx,
      config: this.config,
      targetRole: ctx.targetRole || ctx.resume.meta.targetRole || '',
      fileName: ctx.fileName || ctx.resume.meta.fileName || '',
      attempt: ctx.attempt ?? 1,
    };

    const flags: ValidationFlag[] = [];
    const ruleResults: RuleOutcome[] = [];

    for (const rule of this.rules) {
      const started = Date.now();
      try {
        const produced = await rule.evaluate(fullCtx);
        const normalised = (produced ?? []).map((f) => ({
          ...f,
          // Defensive: honour the rule-level blocking contract.
          blocking: rule.blocking ? f.blocking : false,
          severity: f.severity,
        }));
        flags.push(...normalised);
        ruleResults.push({
          ruleId: rule.id,
          name: rule.name,
          passed: normalised.filter((f) => f.blocking).length === 0,
          flagCount: normalised.length,
          durationMs: Date.now() - started,
        });
      } catch (err) {
        const message = err instanceof Error ? err.message : String(err);
        // A crashed rule must never silently pass a run.
        flags.push({
          ruleId: rule.id,
          code: 'RULE_EXECUTION_ERROR',
          severity: 'error',
          blocking: true,
          message: `Rule ${rule.id} threw: ${message}`,
        });
        ruleResults.push({
          ruleId: rule.id,
          name: rule.name,
          passed: false,
          flagCount: 1,
          durationMs: Date.now() - started,
          error: message,
        });
      }
    }

    return buildResult(fullCtx, flags, ruleResults, this.config);
  }
}

export function buildResult(
  ctx: RuleContext,
  flags: ValidationFlag[],
  ruleResults: RuleOutcome[],
  config: RuleConfig,
): ValidationResult {
  const blockingFailures = flags.filter((f) => f.blocking);
  const warnings = flags.filter((f) => !f.blocking && f.severity === 'warn');
  const infos = flags.filter((f) => f.severity === 'info');
  const attempt = ctx.attempt ?? 1;

  return {
    runId: ctx.resume.meta.runId,
    attempt,
    passed: blockingFailures.length === 0,
    requiresRetry:
      blockingFailures.length > 0 && attempt <= config.maxRetries,
    blockingFailures,
    warnings,
    infos,
    flags,
    ruleResults,
    evaluatedAt: new Date().toISOString(),
  };
}

// packages/rules/src/retry.ts
// Blocking failures trigger a regeneration. Budget: max 2 retries (3 attempts).

import {
  RegenerateFn,
  RetryOutcome,
  RuleContext,
  RuleConfig,
  ValidationFlag,
  ValidationResult,
} from './types';
import { DEFAULT_RULE_CONFIG } from './types';
import { RuleValidator } from './validator';

export interface RunWithRetryOptions {
  validator: RuleValidator;
  initialResume: RuleContext['resume'];
  factBank: RuleContext['factBank'];
  targetRole: string;
  fileName: string;
  regenerate: RegenerateFn;
  config?: RuleConfig;
  llmJudgement?: RuleContext['llmJudgement'];
  /** Invoked after every attempt (attempt is 1-based). */
  onAttempt?: (attempt: number, result: ValidationResult) => void | Promise<void>;
  /** Extra context merged into every RuleContext (e.g. renderedNodes). */
  extra?: Partial<Omit<RuleContext, 'resume' | 'config' | 'attempt'>>;
}

export async function runWithRetry(opts: RunWithRetryOptions): Promise<RetryOutcome> {
  const config = opts.config ?? DEFAULT_RULE_CONFIG;
  const maxAttempts = config.maxRetries + 1; // 1 initial + 2 retries

  let resume = opts.initialResume;
  let result!: ValidationResult;
  let attempt = 1;

  for (; attempt <= maxAttempts; attempt++) {
    result = await opts.validator.validate({
      resume,
      factBank: opts.factBank,
      targetRole: opts.targetRole,
      fileName: opts.fileName,
      llmJudgement: opts.llmJudgement,
      attempt,
      ...opts.extra,
    } as RuleContext);

    await opts.onAttempt?.(attempt, result);

    if (result.blockingFailures.length === 0) {
      return {
        resume,
        result,
        attempts: attempt,
        exhausted: false,
        requiresHumanReview: result.warnings.length > 0 && false,
      };
    }

    if (attempt === maxAttempts) break;

    // Feed blocking failures back to the generator as structured feedback.
    resume = await opts.regenerate(attempt, result.blockingFailures);
  }

  return {
    resume,
    result,
    attempts: maxAttempts,
    exhausted: true,
    requiresHumanReview: true,
  };
}

/** Compact feedback payload handed to the LLM pipeline on retry. */
export function toRetryFeedback(flags: ValidationFlag[]): string {
  return flags
    .map((f) => `[${f.ruleId}/${f.code}] ${f.message}${f.path ? ` @ ${f.path}` : ''}`)
    .join('\n');
}

export const PAGE_WIDTH_TWIPS = 12240;      // US Letter 8.5"
export const PAGE_HEIGHT_TWIPS = 15840;     // US Letter 11"

export const CONTENT_WIDTH_TWIPS = PAGE_WIDTH_TWIPS - MARGIN_TWIPS * 2;

  buffer: Buffer;

const SPACING = {
  sectionBefore: 160,
  sectionAfter: 60,
  itemAfter: 40,
  bulletAfter: 30,
};

function sectionHeading(text: string): Paragraph {
  return new Paragraph({
    spacing: { before: SPACING.sectionBefore, after: SPACING.sectionAfter },
    border: {
      bottom: { style: BorderStyle.SINGLE, size: 4, color: COLOR.rule, space: 2 },
    },
    children: [
      new TextRun({
        text: text.toUpperCase(),
        bold: true,
        size: SIZE.heading,
        color: COLOR.accent,
        font: FONT,
        characterSpacing: 20,
      }),
    ],
  });
}

function bullet(text: string): Paragraph {
  return new Paragraph({
    bullet: { level: 0 },
    spacing: { after: SPACING.bulletAfter },
    children: [new TextRun({ text, size: SIZE.body, font: FONT, color: COLOR.ink })],
  });
}

function body(text: string, opts: { italic?: boolean; color?: string } = {}): Paragraph {
  return new Paragraph({
    spacing: { after: SPACING.itemAfter },
    children: [
      new TextRun({
        text,
        size: SIZE.body,
        font: FONT,
        italics: opts.italic,
        color: opts.color ?? COLOR.ink,
      }),
    ],
  });
}

/** Left/right tab split used for "Title — Company <tab> Dates". */
function tabbedLine(left: TextRun[], right: string): Paragraph {
  return new Paragraph({
    tabStops: [{ type: TabStopType.RIGHT, position: CONTENT_WIDTH_TWIPS }],
    spacing: { before: 60, after: 20 },
    children: [
      ...left,
      new TextRun({ text: '\t', size: SIZE.body, font: FONT }),
      new TextRun({
        text: right, size: SIZE.meta, font: FONT,
        color: COLOR.muted, italics: true,
      }),
    ],
  });
}

function formatRange(start?: string, end?: string | null, current?: boolean): string {
  const fmt = (v?: string | null) => {
    if (!v) return '';
    const m = /^(\d{4})-(\d{2})$/.exec(v);
    if (!m) return v;
    const months = ['Jan','Feb','Mar','Apr','May','Jun',
                    'Jul','Aug','Sep','Oct','Nov','Dec'];
    return `${months[Number(m[2]) - 1]} ${m[1]}`;
  };
  const left = fmt(start);
  const right = current || !end ? 'Present' : fmt(end);
  if (!left && !right) return '';
  return `${left} – ${right}`.trim();
}

function headerSection(doc: ResumeDoc, nodes: RenderNode[]): Paragraph[] {
  const out: Paragraph[] = [];

  out.push(new Paragraph({
    alignment: AlignmentType.CENTER,
    spacing: { after: 20 },
    children: [new TextRun({
      text: doc.contact.name.toUpperCase(),
      bold: true, size: SIZE.name, font: FONT,
      color: COLOR.ink, characterSpacing: 30,
    })],
  }));
  nodes.push({ type: 'heading', path: 'contact.name' });

  const contactBits = [
    doc.contact.email,
    doc.contact.phone,
    doc.contact.location,
    ...(doc.contact.links ?? []),
  ].filter((v): v is string => Boolean(v && v.trim()));

  if (contactBits.length) {
    out.push(new Paragraph({
      alignment: AlignmentType.CENTER,
      spacing: { after: 20 },
      children: [new TextRun({
        text: contactBits.join('  •  '),
        size: SIZE.contact, font: FONT, color: COLOR.muted,
      })],
    }));
    nodes.push({ type: 'paragraph', path: 'contact' });
  }

  if (doc.headline?.trim()) {
    out.push(new Paragraph({
      alignment: AlignmentType.CENTER,
      spacing: { after: 40 },
      children: [new TextRun({
        text: doc.headline.trim(),
        italics: true, size: SIZE.headline, font: FONT, color: COLOR.accent,
      })],
    }));
    nodes.push({ type: 'paragraph', path: 'headline' });
  }

  return out;
}

function summarySection(doc: ResumeDoc, nodes: RenderNode[]): Paragraph[] {
  if (!doc.summary?.trim()) return [];
  nodes.push({ type: 'heading', path: 'summary' });
  nodes.push({ type: 'paragraph', path: 'summary' });
  return [sectionHeading('Summary'), body(doc.summary.trim())];
}

function skillsSection(doc: ResumeDoc, nodes: RenderNode[]): Paragraph[] {
  if (!doc.skills?.length) return [];
  const out: Paragraph[] = [sectionHeading('Core Skills')];
  nodes.push({ type: 'heading', path: 'skills' });

  for (const [i, group] of doc.skills.entries()) {
    const items = (group.items ?? []).filter(Boolean);
    if (!items.length) continue;

    if (group.category) {
      out.push(new Paragraph({
        spacing: { after: SPACING.bulletAfter },
        children: [
          new TextRun({
            text: `${group.category}: `, bold: true,
            size: SIZE.body, font: FONT, color: COLOR.ink,
          }),
          new TextRun({ text: items.join(', '), size: SIZE.body, font: FONT, color: COLOR.ink }),
        ],
      }));
    } else {
      out.push(new Paragraph({
        spacing: { after: SPACING.bulletAfter },
        children: [new TextRun({
          text: items.join('  •  '), size: SIZE.body, font: FONT, color: COLOR.ink,
        })],
      }));
    }
    nodes.push({ type: 'paragraph', path: `skills[${i}]` });
  }
  return out;
}

function experienceSection(doc: ResumeDoc, nodes: RenderNode[]): Paragraph[] {
  if (!doc.experience?.length) return [];
  const out: Paragraph[] = [sectionHeading('Professional Experience')];
  nodes.push({ type: 'heading', path: 'experience' });

  doc.experience.forEach((exp: ExperienceItem, i: number) => {
    out.push(tabbedLine([
      new TextRun({ text: exp.title, bold: true, size: SIZE.body, font: FONT, color: COLOR.ink }),
      new TextRun({
        text: exp.company ? `  —  ${exp.company}` : '',
        size: SIZE.body, font: FONT, color: COLOR.ink,
      }),
      ...(exp.location
        ? [new TextRun({
            text: `  ·  ${exp.location}`,
            size: SIZE.meta, font: FONT, color: COLOR.muted,
          })]
        : []),
    ], formatRange(exp.startDate, exp.endDate, exp.current)));
    nodes.push({ type: 'paragraph', path: `experience[${i}].header` });

    (exp.bullets ?? []).forEach((b, j) => {
      const text = b.trim();
      if (!text) return;
      out.push(bullet(text));
      nodes.push({ type: 'bullet', path: `experience[${i}].bullets[${j}]` });
    });
  });

  return out;
}

function educationSection(doc: ResumeDoc, nodes: RenderNode[]): Paragraph[] {
  if (!doc.education?.length) return [];
  const out: Paragraph[] = [sectionHeading('Education')];
  nodes.push({ type: 'heading', path: 'education' });

  doc.education.forEach((edu: EducationItem, i: number) => {
    const degree = [edu.degree, edu.field].filter(Boolean).join(', ');
    out.push(tabbedLine([
      new TextRun({ text: degree || edu.institution, bold: true,
        size: SIZE.body, font: FONT, color: COLOR.ink }),
      ...(degree && edu.institution
        ? [new TextRun({ text: `  —  ${edu.institution}`, size: SIZE.body,
            font: FONT, color: COLOR.ink })]
        : []),
    ], formatRange(edu.startDate, edu.endDate)));
    nodes.push({ type: 'paragraph', path: `education[${i}].header` });

    if (edu.grade?.trim()) {
      out.push(new Paragraph({
        spacing: { after: SPACING.itemAfter },
        children: [new TextRun({
          text: edu.grade.trim(), size: SIZE.meta, font: FONT, color: COLOR.muted,
        })],
      }));
      nodes.push({ type: 'paragraph', path: `education[${i}].grade` });
    }
  });

  return out;
}

function projectsSection(doc: ResumeDoc, nodes: RenderNode[]): Paragraph[] {
  if (!doc.projects?.length) return [];
  const out: Paragraph[] = [sectionHeading('Selected Projects')];
  nodes.push({ type: 'heading', path: 'projects' });

  doc.projects.forEach((p, i) => {
    out.push(tabbedLine([
      new TextRun({ text: p.name, bold: true, size: SIZE.body, font: FONT, color: COLOR.ink }),
    ], p.link ?? ''));
    nodes.push({ type: 'paragraph', path: `projects[${i}].header` });

    if (p.description?.trim()) {
      out.push(body(p.description.trim()));
      nodes.push({ type: 'paragraph', path: `projects[${i}].description` });
    }
    (p.bullets ?? []).forEach((b, j) => {
      if (!b.trim()) return;
      out.push(bullet(b.trim()));
      nodes.push({ type: 'bullet', path: `projects[${i}].bullets[${j}]` });
    });
  });

  return out;
}

function inlineListSection(
  title: string,
  items: string[] | undefined,
  path: string,
  nodes: RenderNode[],
): Paragraph[] {
  const clean = (items ?? []).map((s) => s.trim()).filter(Boolean);
  if (!clean.length) return [];
  nodes.push({ type: 'heading', path });
  nodes.push({ type: 'paragraph', path });
  return [
    sectionHeading(title),
    new Paragraph({
      spacing: { after: SPACING.itemAfter },
      children: [new TextRun({
        text: clean.join('  •  '), size: SIZE.body, font: FONT, color: COLOR.ink,
      })],
    }),
  ];
}

export async function renderResumeDocx(doc: ResumeDoc): Promise<RenderedDocx> {
  const nodes: RenderNode[] = [];

  const children: Paragraph[] = [
    ...headerSection(doc, nodes),
    ...summarySection(doc, nodes),
    ...skillsSection(doc, nodes),
    ...experienceSection(doc, nodes),
    ...educationSection(doc, nodes),
    ...projectsSection(doc, nodes),
    ...inlineListSection('Certifications', doc.certifications, 'certifications', nodes),
    ...inlineListSection('Languages', doc.languages, 'languages', nodes),
  ];

  const document = new Document({
    creator: 'Resume Engine',
    title: `${doc.contact.name} — ${doc.meta.targetRole}`,
    description: `Generated for run ${doc.meta.runId}`,
    styles: {
      default: {
        document: {
          run: { font: FONT, size: SIZE.body, color: COLOR.ink },
        },
      },
    },
    numbering: {
      config: [{
        reference: 'resume-bullets',
        levels: [{
          level: 0,
          format: LevelFormat.BULLET,
          text: '•',
          alignment: AlignmentType.LEFT,
          style: { paragraph: { indent: { left: convertInchesToTwip(0.22),
                                          hanging: convertInchesToTwip(0.16) } } },
        }],
      }],
    },
    sections: [{
      properties: {
        page: {
          size: { width: PAGE_WIDTH_TWIPS, height: PAGE_HEIGHT_TWIPS },
          margin: {
            top: MARGIN_TWIPS, right: MARGIN_TWIPS,
            bottom: MARGIN_TWIPS, left: MARGIN_TWIPS,
          },
        },
      },
      children,
    }],
  });

  const buffer = await Packer.toBuffer(document);
  return { buffer, nodes };
}

// packages/qa/src/pdf.ts
// Stage 8 — LibreOffice headless DOCX -> PDF conversion, page counting, QA.

import { execFile } from 'node:child_process';
import { mkdtemp, readFile, rm, writeFile, access } from 'node:fs/promises';
import { constants as FS } from 'node:fs';
import { tmpdir } from 'node:os';
import { join, basename, parse } from 'node:path';
import { promisify } from 'node:util';
import { PDFDocument } from 'pdf-lib';
import type { ValidationFlag } from '@resume-engine/rules';

const execFileAsync = promisify(execFile);

export interface PdfQaResult {
  pdf: Buffer;
  pageCount: number;
  overflow: boolean;
  flags: ValidationFlag[];
  conversionMs: number;
}

export interface ConvertOptions {
  /** Path to soffice binary. */
  sofficePath?: string;
  timeoutMs?: number;
  maxPages?: number;
  runId?: string;
}

/** Locates soffice across common Linux/macOS install locations. */
export async function resolveSoffice(explicit?: string): Promise<string> {
  const candidates = [
    explicit,
    process.env.SOFFICE_PATH,
    '/usr/bin/soffice',
    '/usr/local/bin/soffice',
    '/opt/libreoffice/program/soffice',
    '/Applications/LibreOffice.app/Contents/MacOS/soffice',
  ].filter((c): c is string => Boolean(c));

  for (const c of candidates) {
    try {
      await access(c, FS.X_OK);
      return c;
    } catch { /* try next */ }
  }
  throw new Error(
    'LibreOffice (soffice) not found. Set SOFFICE_PATH or install libreoffice-core.',
  );
}

/**
 * Converts a DOCX buffer to PDF in an isolated temp directory.
 * `--headless --norestore` keeps it safe inside containers.
 */
export async function convertDocxToPdf(
  docx: Buffer,
  opts: ConvertOptions = {},
): Promise<{ pdf: Buffer; conversionMs: number }> {
  const soffice = await resolveSoffice(opts.sofficePath);
  const timeout = opts.timeoutMs ?? 120_000;
  const dir = await mkdtemp(join(tmpdir(), 'resume-qa-'));

  try {
    const docxPath = join(dir, 'input.docx');
    await writeFile(docxPath, docx);

    const started = Date.now();
    await execFileAsync(
      soffice,
      [
        '--headless',
        '--norestore',
        '--nologo',
        '--nolockcheck',
        '--nodefault',
        '--nofirststartwizard',
        `-env:UserInstallation=file://${join(dir, 'lo-profile')}`,
        '--convert-to', 'pdf:writer_pdf_Export',
        '--outdir', dir,
        docxPath,
      ],
      { timeout, maxBuffer: 16 * 1024 * 1024 },
    );
    const conversionMs = Date.now() - started;

    const pdfPath = join(dir, `${parse(basename(docxPath)).name}.pdf`);
    const pdf = await readFile(pdfPath);
    return { pdf, conversionMs };
  } finally {
    await rm(dir, { recursive: true, force: true }).catch(() => { /* noop */ });
  }
}

export async function countPdfPages(pdf: Buffer): Promise<number> {
  const doc = await PDFDocument.load(pdf, { ignoreEncryption: true });
  return doc.getPageCount();
}

/**
 * Full Stage 8 QA: convert, count pages, emit PAGE_OVERFLOW when over budget.
 * PAGE_OVERFLOW is a warning by design (never blocks the run).
 */
export async function runPdfQa(
  docx: Buffer,
  opts: ConvertOptions = {},
): Promise<PdfQaResult> {
  const maxPages = opts.maxPages ?? 3;
  const { pdf, conversionMs } = await convertDocxToPdf(docx, opts);
  const pageCount = await countPdfPages(pdf);
  const overflow = pageCount > maxPages;

  const flags: ValidationFlag[] = [];
  if (overflow) {
    flags.push({
      ruleId: 'R15',
      code: 'PAGE_OVERFLOW',
      severity: 'warn',
      blocking: false,
      message: `Rendered PDF is ${pageCount} pages; limit is ${maxPages}.`,
      path: undefined,
      meta: { pageCount, maxPages, runId: opts.runId, conversionMs },
    });
  }

  return { pdf, pageCount, overflow, flags, conversionMs };
}

// packages/storage/src/artifact-store.ts
// S3 / Cloudflare R2 artifact storage with presigned GET URLs.

import {
  S3Client,
  PutObjectCommand,
  GetObjectCommand,
  HeadObjectCommand,
} from '@aws-sdk/client-s3';
import { getSignedUrl } from '@aws-sdk/s3-request-presigner';

export interface ArtifactStoreConfig {
  bucket: string;
  region?: string;
  endpoint?: string;        // R2: https://<account>.r2.cloudflarestorage.com
  accessKeyId: string;
  secretAccessKey: string;
  forcePathStyle?: boolean;
  signedUrlTtlSeconds?: number;
}

export interface ArtifactRef {
  key: string;
  bucket: string;
  contentType: string;
  size: number;
  url: string;              // presigned
  expiresAt: string;
}

export class ArtifactStore {
  private readonly client: S3Client;
  private readonly ttl: number;

  constructor(private readonly cfg: ArtifactStoreConfig) {
    this.client = new S3Client({
      region: cfg.region ?? 'auto',
      endpoint: cfg.endpoint,
      forcePathStyle: cfg.forcePathStyle ?? Boolean(cfg.endpoint),
      credentials: {
        accessKeyId: cfg.accessKeyId,
        secretAccessKey: cfg.secretAccessKey,
      },
    });
    this.ttl = cfg.signedUrlTtlSeconds ?? 900; // 15 min
  }

  async put(
    key: string,
    body: Buffer,
    contentType: string,
    metadata?: Record<string, string>,
  ): Promise<ArtifactRef> {
    await this.client.send(new PutObjectCommand({
      Bucket: this.cfg.bucket,
      Key: key,
      Body: body,
      ContentType: contentType,
      Metadata: metadata,
    }));
    return this.ref(key, contentType, body.byteLength);
  }

  async signedGetUrl(key: string, ttlSeconds?: number): Promise<string> {
    const ttl = ttlSeconds ?? this.ttl;
    return getSignedUrl(
      this.client,
      new GetObjectCommand({ Bucket: this.cfg.bucket, Key: key }),
      { expiresIn: ttl },
    );
  }

  async exists(key: string): Promise<boolean> {
    try {
      await this.client.send(
        new HeadObjectCommand({ Bucket: this.cfg.bucket, Key: key }),
      );
      return true;
    } catch {
      return false;
    }
  }

  async ref(key: string, contentType: string, size = 0): Promise<ArtifactRef> {
    const url = await this.signedGetUrl(key);
    return {
      key,
      bucket: this.cfg.bucket,
      contentType,
      size,
      url,
      expiresAt: new Date(Date.now() + this.ttl * 1000).toISOString(),
    };
  }
}

// apps/api/src/routes/runs.ts
// Stage 8 endpoints:
//   GET /runs/:id/output
//   GET /runs/:id/validation
//   GET /runs/:id/docx   -> signed S3/R2 URL
//   GET /runs/:id/pdf    -> signed S3/R2 URL

import { Router, Request, Response, NextFunction } from 'express';
import type { ValidationResult } from '@resume-engine/rules';
import type { ArtifactStore } from '@resume-engine/storage';
import { artifactKeys } from '@resume-engine/storage';

export interface RunArtifacts {
  docxKey?: string;
  pdfKey?: string;
  docxSize?: number;
  pdfSize?: number;
  pageCount?: number;
  generatedAt?: string;
}

export interface RunRecord {
  id: string;
  status: 'queued' | 'generating' | 'validating' | 'rendering'
        | 'exporting' | 'succeeded' | 'failed' | 'needs_review';
  attempts: number;
  fileName: string;
  slug: string;
  targetRole: string;
  createdAt: string;
  updatedAt: string;
  artifacts: RunArtifacts;
  validation?: ValidationResult;
  error?: { code: string; message: string } | null;
  ownerId: string;
}

export interface RunRepository {
  findById(id: string): Promise<RunRecord | null>;
}

export interface RunsRouterDeps {
  runs: RunRepository;
  store: ArtifactStore;
  /** 404/500 helper kept explicit for testability. */
  logger?: { error: (msg: string, meta?: unknown) => void };
}

class HttpError extends Error {
  constructor(public status: number, message: string, public code = 'ERROR') {
    super(message);
  }
}

export function createRunsRouter(deps: RunsRouterDeps): Router {
  const router = Router();
  const { runs, store, logger } = deps;

  async function loadRun(req: Request, res: Response): Promise<RunRecord | null> {
    const { id } = req.params;
    if (!/^[A-Za-z0-9_-]{6,64}$/.test(id)) {
      res.status(400).json({ error: { code: 'INVALID_RUN_ID', message: 'Bad run id.' } });
      return null;
    }
    const run = await runs.findById(id);
    if (!run) {
      res.status(404).json({ error: { code: 'RUN_NOT_FOUND', message: `Run ${id} not found.` } });
      return null;
    }
    if (!req.auth?.userId || req.auth.userId !== run.ownerId) {
      res.status(403).json({ error: { code: 'FORBIDDEN', message: 'Not your run.' } });
      return null;
    }
    return run;
  }

  /* ------------------------------ GET /output ------------------------------ */

  router.get('/:id/output', async (req, res, next) => {
    try {
      const run = await loadRun(req, res);
      if (!run) return;

      const keys = artifactKeys(run.id, run.slug);
      const [docx, pdf] = await Promise.all([
        run.artifacts.docxKey
          ? store.ref(run.artifacts.docxKey, 'application/vnd.openxmlformats-officedocument.wordprocessingml.document', run.artifacts.docxSize ?? 0)
          : null,
        run.artifacts.pdfKey
          ? store.ref(run.artifacts.pdfKey, 'application/pdf', run.artifacts.pdfSize ?? 0)
          : null,
      ]);

      res.json({
        runId: run.id,
        status: run.status,
        attempts: run.attempts,
        fileName: run.fileName,
        targetRole: run.targetRole,
        generatedAt: run.artifacts.generatedAt ?? run.updatedAt,
        pageCount: run.artifacts.pageCount ?? null,
        validationSummary: run.validation
          ? {
              passed: run.validation.passed,
              blocking: run.validation.blockingFailures.length,
              warnings: run.validation.warnings.length,
              attempts: run.validation.attempt,
            }
          : null,
        artifacts: {
          docx: docx ?? null,
          pdf: pdf ?? null,
          validation: `${req.baseUrl}/${run.id}/validation`,
        },
        keys,
        error: run.error ?? null,
      });
    } catch (err) {
      next(err);
    }
  });

  /* ---------------------------- GET /validation ---------------------------- */

  router.get('/:id/validation', async (req, res, next) => {
    try {
      const run = await loadRun(req, res);
      if (!run) return;

      if (!run.validation) {
        res.status(409).json({
          error: {
            code: 'VALIDATION_NOT_READY',
            message: 'Validation has not completed for this run yet.',
          },
          status: run.status,
        });
        return;
      }

      const v = run.validation;
      const includeRules = req.query.rules === '1' || req.query.rules === 'true';
      const severity = String(req.query.severity ?? '').toLowerCase();

      let flags = v.flags;
      if (['error', 'warn', 'info'].includes(severity)) {
        flags = flags.filter((f) => f.severity === severity);
      }
      if (req.query.rule) {
        const rid = String(req.query.rule).toUpperCase();
        flags = flags.filter((f) => f.ruleId === rid);
      }

      res.json({
        runId: run.id,
        attempt: v.attempt,
        passed: v.passed,
        requiresRetry: v.requiresRetry,
        evaluatedAt: v.evaluatedAt,
        counts: {
          total: v.flags.length,
          blocking: v.blockingFailures.length,
          warnings: v.warnings.length,
          infos: v.infos.length,
        },
        blockingFailures: v.blockingFailures,
        warnings: v.warnings,
        infos: v.infos,
        flags,
        ...(includeRules ? { ruleResults: v.ruleResults } : {}),
      });
    } catch (err) {
      next(err);
    }
  });

  /* ------------------------------ GET /docx -------------------------------- */

  router.get('/:id/docx', async (req, res, next) => {
    try {
      const run = await loadRun(req, res);
      if (!run) return;

      if (!run.artifacts.docxKey) {
        res.status(409).json({
          error: { code: 'DOCX_NOT_READY', message: 'DOCX artifact is not available.' },
          status: run.status,
        });
        return;
      }

      const ref = await store.ref(
        run.artifacts.docxKey,
        'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
        run.artifacts.docxSize ?? 0,
      );

      if (req.query.redirect === '1') {
        res.redirect(302, ref.url);
        return;
      }
      res.setHeader('Cache-Control', 'no-store');
      res.json(ref);
    } catch (err) {
      next(err);
    }
  });

  /* ------------------------------- GET /pdf -------------------------------- */

  router.get('/:id/pdf', async (req, res, next) => {
    try {
      const run = await loadRun(req, res);
      if (!run) return;

      if (!run.artifacts.pdfKey) {
        res.status(409).json({
          error: { code: 'PDF_NOT_READY', message: 'PDF artifact is not available.' },
          status: run.status,
        });
        return;
      }

      const ref = await store.ref(
        run.artifacts.pdfKey, 'application/pdf', run.artifacts.pdfSize ?? 0,
      );

      if (req.query.redirect === '1') {
        res.redirect(302, ref.url);
        return;
      }
      res.setHeader('Cache-Control', 'no-store');
      res.json(ref);
    } catch (err) {
      next(err);
    }
  });

  /* ------------------------------- error sink ------------------------------ */

  router.use((err: unknown, _req: Request, res: Response, _next: NextFunction) => {
    if (err instanceof HttpError) {
      res.status(err.status).json({ error: { code: err.code, message: err.message } });
      return;
    }
    logger?.error('runs route failure', err);
    res.status(500).json({
      error: { code: 'INTERNAL_ERROR', message: 'Unexpected error.' },
    });
  });

  return router;
}

// apps/api/src/routes/admin-2fa.ts
// Victor sign-off item 5: enrollment is POST /admin/2fa/enroll (not GET).
// Victor sign-off item 6: the bootstrap token arrives in X-Bootstrap-Token.

import { Router, Request, Response, NextFunction, RequestHandler } from 'express';
import { timingSafeEqual } from 'node:crypto';
import { authenticator } from 'otplib';
import { toDataURL } from 'qrcode';

export interface TwoFaStore {
  findAdmin(adminId: string): Promise<{ id: string; email: string; totpSecret?: string | null } | null>;
  saveEnrollment(adminId: string, secret: string, recoveryCodes: string[]): Promise<void>;
}

export interface Admin2faDeps {
  admins: TwoFaStore;
  bootstrapToken: string;
  issuer?: string;
}

function safeEqual(a: string, b: string): boolean {
  const ba = Buffer.from(a);
  const bb = Buffer.from(b);
  if (ba.length !== bb.length) return false;
  return timingSafeEqual(ba, bb);
}

/** Header-based bootstrap authentication for first-run 2FA enrollment. */
export function requireBootstrapToken(expected: string): RequestHandler {
  return (req: Request, res: Response, next: NextFunction) => {
    const provided = req.header('X-Bootstrap-Token');
    if (!provided || !safeEqual(provided, expected)) {
      res.status(401).json({
        error: {
          code: 'INVALID_BOOTSTRAP_TOKEN',
          message: 'X-Bootstrap-Token header missing or invalid.',
        },
      });
      return;
    }
    next();
  };
}

function generateRecoveryCodes(count = 8): string[] {
  const out: string[] = [];
  const alphabet = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  for (let i = 0; i < count; i++) {
    let code = '';
    for (let j = 0; j < 10; j++) {
      if (j === 5) code += '-';
      code += alphabet[Math.floor(Math.random() * alphabet.length)];
    }
    out.push(code);
  }
  return out;
}

export function createAdmin2faRouter(deps: Admin2faDeps): Router {
  const router = Router();
  const issuer = deps.issuer ?? 'Resume Engine';

  /** POST /admin/2fa/enroll */
  router.post(
    '/2fa/enroll',
    requireBootstrapToken(deps.bootstrapToken),
    async (req: Request, res: Response, next: NextFunction) => {
      try {
        const adminId = String(req.body?.adminId ?? '');
        if (!adminId) {
          res.status(400).json({
            error: { code: 'ADMIN_ID_REQUIRED', message: 'adminId is required.' },
          });
          return;
        }

        const admin = await deps.admins.findAdmin(adminId);
        if (!admin) {
          res.status(404).json({
            error: { code: 'ADMIN_NOT_FOUND', message: `Admin ${adminId} not found.` },
          });
          return;
        }
        if (admin.totpSecret) {
          res.status(409).json({
            error: {
              code: 'ALREADY_ENROLLED',
              message: '2FA is already enrolled for this admin.',
            },
          });
          return;
        }

        const secret = authenticator.generateSecret();
        const otpauth = authenticator.keyuri(admin.email, issuer, secret);
        const qrDataUrl = await toDataURL(otpauth);
        const recoveryCodes = generateRecoveryCodes();

        await deps.admins.saveEnrollment(admin.id, secret, recoveryCodes);

        res.status(201).json({
          adminId: admin.id,
          issuer,
          account: admin.email,
          secret,
          otpauth,
          qrDataUrl,
          recoveryCodes,
          note: 'Store recovery codes offline. Secret is shown once.',
        });
      } catch (err) {
        next(err);
      }
    },
  );

  return router;
}

// apps/api/src/pipeline/stage-6-7-8.ts
// Wires Stage 6 (validate) -> Stage 7 (render DOCX) -> Stage 8 (PDF + QA + upload).

import {
  RuleValidator,
  RuleConfig,
  DEFAULT_RULE_CONFIG,
  ResumeDoc,
  FactBankEntry,
  ValidationResult,
  RegenerateFn,
  runWithRetry,
  LlmJudgement,
} from '@resume-engine/rules';
import { renderResumeDocx } from '@resume-engine/renderer';
import { runPdfQa } from '@resume-engine/qa';
import { ArtifactStore, artifactKeys, slugifyFileName } from '@resume-engine/storage';

export interface PipelineDeps {
  validator?: RuleValidator;
  config?: RuleConfig;
  store: ArtifactStore;
  regenerate: RegenerateFn;
  onValidation?: (runId: string, result: ValidationResult) => Promise<void>;
  onArtifacts?: (runId: string, artifacts: {
    docxKey: string; pdfKey: string; pageCount: number;
    docxSize: number; pdfSize: number;
  }) => Promise<void>;
}

export interface Stage6to8Input {
  runId: string;
  resume: ResumeDoc;
  factBank: FactBankEntry[];
  targetRole: string;
  fileName: string;
  llmJudgement?: LlmJudgement;
}

export interface Stage6to8Output {
  validation: ValidationResult;
  attempts: number;
  requiresHumanReview: boolean;
  docxKey: string | null;
  pdfKey: string | null;
  pageCount: number | null;
  postRenderValidation: ValidationResult;
}

export async function runStages6to8(
  input: Stage6to8Input,
  deps: PipelineDeps,
): Promise<Stage6to8Output> {
  const config = deps.config ?? DEFAULT_RULE_CONFIG;
  const validator = deps.validator ?? new RuleValidator(config);

  /* ------------------------------ Stage 6 ------------------------------- */

  const retry = await runWithRetry({
    validator,
    initialResume: input.resume,
    factBank: input.factBank,
    targetRole: input.targetRole,
    fileName: input.fileName,
    llmJudgement: input.llmJudgement,
    regenerate: deps.regenerate,
    config,
    onAttempt: async (_attempt, result) => {
      await deps.onValidation?.(input.runId, result);
    },
  });

  if (retry.exhausted) {
    // Blocking failures survived the retry budget — do not render.
    return {
      validation: retry.result,
      attempts: retry.attempts,
      requiresHumanReview: true,
      docxKey: null,
      pdfKey: null,
      pageCount: null,
      postRenderValidation: retry.result,
    };
  }

  /* ------------------------------ Stage 7 ------------------------------- */

  const resume = retry.resume;
  const { buffer: docx, nodes } = await renderResumeDocx(resume);

  /* ------------------------------ Stage 8 ------------------------------- */

  const qa = await runPdfQa(docx, {
    maxPages: config.maxPages,
    runId: input.runId,
  });

  // Post-render validation pass: R13 (node inventory) + R15 (page count).
  const postRenderValidation = await validator.validate({
    resume,
    factBank: input.factBank,
    targetRole: input.targetRole,
    fileName: input.fileName,
    llmJudgement: input.llmJudgement,
    pageCount: qa.pageCount,
    renderedNodes: nodes,
    attempt: retry.attempts,
  });

  /* ------------------------------ Upload -------------------------------- */

  const slug = slugifyFileName(input.fileName);
  const keys = artifactKeys(input.runId, slug);

  const [docxRef, pdfRef] = await Promise.all([
    deps.store.put(keys.docx, docx,
      'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
      { runId: input.runId, attempt: String(retry.attempts) }),
    deps.store.put(keys.pdf, qa.pdf, 'application/pdf',
      { runId: input.runId, pages: String(qa.pageCount) }),
  ]);

  await deps.store.put(
    keys.validation,
    Buffer.from(JSON.stringify({
      stage6: retry.result,
      stage8: postRenderValidation,
      pageCount: qa.pageCount,
      overflow: qa.overflow,
    }, null, 2)),
    'application/json',
  );

  await deps.onArtifacts?.(input.runId, {
    docxKey: docxRef.key,
    pdfKey: pdfRef.key,
    pageCount: qa.pageCount,
    docxSize: docxRef.size,
    pdfSize: pdfRef.size,
  });

  return {
    validation: retry.result,
    attempts: retry.attempts,
    requiresHumanReview: false,
    docxKey: docxRef.key,
    pdfKey: pdfRef.key,
    pageCount: qa.pageCount,
    postRenderValidation,
  };
}

================================================================================
 SECTION Q — IMPLEMENTATION NOTES & DEVIATIONS
================================================================================

1. R2 close-variant metric
   Implemented as the overlap coefficient (|A ∩ B| / min(|A|,|B|)) over
   lower-cased tokens with `roleStopWords` removed, threshold 0.70.
   Chosen over Jaccard so that "Data Scientist" vs "Lead Data Scientist"
   scores 1.00 (the intent of "ignoring stop words like Senior, Lead").
   Flip to Jaccard by editing `tokenOverlap` in util/text.ts if the master
   spec pins the other definition.

2. R9 / R10 non-blocking contract
   Enforced twice: at rule level (blocking:false) and again inside
   RuleValidator.validate(), which coerces `blocking` to false for any rule
   declared non-blocking. A future rule edit cannot accidentally make the
   run retry on an LLM judgement.

3. R20 retry semantics
   runWithRetry() allows 1 initial attempt + config.maxRetries (2) = 3 total.
   If blocking failures survive the budget, Stage 7/8 are skipped entirely
   and the run lands in `needs_review` rather than shipping an ungrounded PDF.

4. R13 enforcement point
   The DOCX renderer emits no tables/images/textboxes by construction, so
   pre-render R13 is a no-op. The renderer returns a `RenderNode[]` inventory
   that is fed back into the post-render validation pass, giving a positive
   assertion rather than an absence-of-evidence check.

5. R15 / PAGE_OVERFLOW
   Both runPdfQa() and rule R15 emit `PAGE_OVERFLOW` for a >3-page PDF.
   They are deduplicated by `code` when persisted; the Stage 8 flag carries
   the conversion timing in `meta`, R15 carries the rule attribution.

6. LibreOffice isolation
   Each conversion uses a fresh temp dir plus a per-run
   `-env:UserInstallation` profile, so concurrent conversions cannot collide
   on the LibreOffice user profile lock.

7. Filename / slug
   R18 validates the *human* filename (`Jane Doe_CTO_Acme.docx`, spaces
   allowed by Victor's regex). Storage keys use `slugifyFileName()` which
   swaps spaces for hyphens — the regex is intentionally not applied to
   object keys.

8. Auth on /runs/:id/*
   Routes assume `req.auth.userId` is populated by the existing auth
   middleware and enforce `run.ownerId === req.auth.userId`. Signed artifact
   URLs are short-lived (15 min default) and never cached (`no-store`).

9. GET vs POST
   /runs/:id/docx and /runs/:id/pdf return a JSON `ArtifactRef` by default
   and 302 to the signed URL when `?redirect=1`. This keeps the mobile
   client able to refresh URLs without a second round-trip.

10. Placeholder sections
    `01-SPEC-v3.md` and `02-RESUME_ENGINE_SPEC.md` arrived as literal paste
    placeholders. If the real ruleset differs on any of the twenty rules,
    only `packages/rules/src/rules/index.ts` and `DEFAULT_RULE_CONFIG` in
    `types.ts` need to change — no downstream package couples to a rule body.
