import { NestFactory } from '@nestjs/core';
import { ValidationPipe } from '@nestjs/common';
import { AppModule } from './app.module';
import { PrismaService } from './prisma/prisma.service';
import { ensureAdmin } from './auth/bootstrap-admin';
import { AllExceptionsFilter } from './common/errors/all-exceptions.filter';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);
  app.setGlobalPrefix('v1');
  app.useGlobalPipes(
    new ValidationPipe({
      whitelist: true,
      transform: true,
      forbidNonWhitelisted: false,
    }),
  );
  app.useGlobalFilters(new AllExceptionsFilter());
  app.enableCors({ origin: true, credentials: true });
  app.enableShutdownHooks();

  await app.init();

  const prisma = app.get(PrismaService);
  await ensureAdmin(prisma);

  const port = process.env.PORT ? Number(process.env.PORT) : 3000;
  await app.listen(port);
}

bootstrap();

import {
  ArgumentsHost,
  Catch,
  ExceptionFilter,
  HttpException,
} from '@nestjs/common';
import type { Response } from 'express';
import { AppError } from './app-error';

@Catch()
export class AllExceptionsFilter implements ExceptionFilter {
  catch(exception: unknown, host: ArgumentsHost): void {
    const ctx = host.switchToHttp();
    const res = ctx.getResponse<Response>();

    if (exception instanceof AppError) {
      res.status(exception.status).json({
        error: { code: exception.code, message: exception.message },
      });
      return;
    }

    if (exception instanceof HttpException) {
      const status = exception.getStatus();
      const response = exception.getResponse();
      const message =
        typeof response === 'string'
          ? response
          : ((response as { message?: string | string[] }).message ?? 'Request failed');
      res.status(status).json({ error: { code: 'HTTP_ERROR', message } });
      return;
    }

    // Never log request bodies or resume text.
    console.error('[unhandled]', exception instanceof Error ? exception.message : exception);
    res.status(500).json({
      error: { code: 'INTERNAL_ERROR', message: 'Internal server error' },
    });
  }
}

import { Injectable, OnModuleDestroy } from '@nestjs/common';
import Redis from 'ioredis';

@Injectable()
export class RedisService implements OnModuleDestroy {
  private readonly client: Redis | null = null;

  constructor() {
    const url = process.env.REDIS_URL;
    if (!url) return;

    this.client = new Redis(url, {
      lazyConnect: true,
      maxRetriesPerRequest: 1,
      enableOfflineQueue: false,
    });
    this.client.on('error', () => undefined);
    this.client.connect().catch(() => undefined);
  }

  async incrWithTtl(key: string, ttlSeconds: number): Promise<number | null> {
    if (!this.client || this.client.status !== 'ready') return null;
    try {
      const value = await this.client.incr(key);
      if (value === 1) await this.client.expire(key, ttlSeconds);
      return value;
    } catch {
      return null;
    }
  }

  async onModuleDestroy(): Promise<void> {
    if (this.client) {
      await this.client.quit().catch(() => undefined);
    }
  }
}

import { Injectable } from '@nestjs/common';
import { AppError } from '../errors/app-error';
import { RedisService } from './redis.service';

@Injectable()
export class RateLimitService {
  constructor(private readonly redis: RedisService) {}

  async consume(key: string, limit: number, ttlSeconds: number): Promise<void> {
    const count = await this.redis.incrWithTtl(key, ttlSeconds);
    if (count !== null && count > limit) {
      throw new AppError('RATE_LIMITED', 429, 'Too many attempts. Try again later.');
    }
  }
}

import * as argon2 from 'argon2';
import { createHash } from 'crypto';

export const ARGON2_OPTIONS: argon2.Options = {
  type: argon2.argon2id,
  memoryCost: 65536, // 64 MB
  timeCost: 3,
  parallelism: 4,
};

export const ACCESS_TOKEN_TTL_SECONDS = 15 * 60;
export const REFRESH_TOKEN_TTL_DAYS = 30;

export function sha256(input: string): string {
  return createHash('sha256').update(input).digest('hex');
}

export function assertPasswordPolicy(password: string): void {
  const ok =
    password.length >= 12 &&
    /[A-Z]/.test(password) &&
    /[a-z]/.test(password) &&
    /[0-9]/.test(password) &&
    /[^A-Za-z0-9]/.test(password);
  if (!ok) {
    throw new Error(
      'Password must be at least 12 characters and contain upper, lower, digit and symbol.',
    );
  }
}

import type { PrismaClient } from '@prisma/client';
import * as argon2 from 'argon2';
import { ARGON2_OPTIONS } from './auth.constants';

export async function ensureAdmin(prisma: PrismaClient): Promise<void> {
  const email = process.env.ADMIN_EMAIL;
  if (!email) return;

  const existing = await prisma.user.findUnique({ where: { email } });

  if (existing) {
    if (!existing.isAdmin) {
      await prisma.user.update({
        where: { id: existing.id },
        data: { isAdmin: true, adminUnlimited: true, role: 'ADMIN' },
      });
    }
    return;
  }

  const password = process.env.ADMIN_SEED_PASSWORD;
  if (!password) throw new Error('ADMIN_SEED_PASSWORD required on first boot');

  await prisma.user.create({
    data: {
      email,
      passwordHash: await argon2.hash(password, ARGON2_OPTIONS),
      firstName: 'Victor',
      lastName: 'Igboekwe',
      role: 'ADMIN',
      isAdmin: true,
      adminUnlimited: true,
      emailVerified: true,
      mustChangePassword: process.env.NODE_ENV === 'production',
    },
  });
}

import { Injectable } from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import * as argon2 from 'argon2';
import { randomBytes, randomUUID } from 'crypto';
import { PrismaService } from '../prisma/prisma.service';
import { AppError } from '../common/errors/app-error';
import { RateLimitService } from '../common/rate-limit/rate-limit.service';
import { LoginDto } from './dto/login.dto';
import { ensureAdmin } from './bootstrap-admin';
import {
  ACCESS_TOKEN_TTL_SECONDS,
  REFRESH_TOKEN_TTL_DAYS,
  sha256,
} from './auth.constants';

@Injectable()
export class AuthService {
  constructor(
    private readonly prisma: PrismaService,
    private readonly jwt: JwtService,
    private readonly rateLimit: RateLimitService,
  ) {}

  async bootstrapAdmin(token?: string) {
    if (!token) {
      throw new AppError(
        'BOOTSTRAP_TOKEN_REQUIRED',
        401,
        'X-Bootstrap-Token header is required.',
      );
    }

    const count = await this.prisma.user.count();
    const expected = process.env.ADMIN_BOOTSTRAP_TOKEN;

    // Idempotent rule: succeeds when there are no users OR the token matches.
    if (count > 0 && (!expected || token !== expected)) {
      throw new AppError('FORBIDDEN', 403, 'Invalid bootstrap token.');
    }

    if (!process.env.ADMIN_EMAIL) {
      throw new AppError(
        'ADMIN_NOT_CONFIGURED',
        500,
        'ADMIN_EMAIL is not configured.',
      );
    }

    await ensureAdmin(this.prisma);
    return { ok: true, hadUsers: count > 0 };
  }

  async login(dto: LoginDto, ip?: string, userAgent?: string) {
    const user = await this.prisma.user.findUnique({
      where: { email: dto.email },
    });

    // Admin: 5 attempts / 15 min / IP. Standard: 10 attempts / 15 min / IP.
    await this.rateLimit.consume(
      `login:${ip ?? 'unknown'}`,
      user?.isAdmin ? 5 : 10,
      15 * 60,
    );

    if (!user) {
      throw new AppError('INVALID_CREDENTIALS', 401, 'Invalid email or password.');
    }

    const valid = await argon2.verify(user.passwordHash, dto.password);
    if (!valid) {
      throw new AppError('INVALID_CREDENTIALS', 401, 'Invalid email or password.');
    }

    const jti = randomUUID();
    const accessToken = await this.jwt.signAsync(
      {
        sub: user.id,
        role: user.role,
        isAdmin: user.isAdmin,
        jti,
      },
      { expiresIn: ACCESS_TOKEN_TTL_SECONDS },
    );

    const refreshToken = randomBytes(32).toString('base64url');
    await this.prisma.session.create({
      data: {
        userId: user.id,
        refreshHash: sha256(refreshToken),
        userAgent,
        ip,
        expiresAt: new Date(
          Date.now() + REFRESH_TOKEN_TTL_DAYS * 24 * 60 * 60 * 1000,
        ),
      },
    });

    return {
      accessToken,
      refreshToken,
      tokenType: 'Bearer',
      expiresIn: ACCESS_TOKEN_TTL_SECONDS,
    };
  }

  async me(userId: string) {
    const user = await this.prisma.user.findUnique({
      where: { id: userId },
      select: {
        id: true,
        email: true,
        role: true,
        isAdmin: true,
        adminUnlimited: true,
        emailVerified: true,
        mustChangePassword: true,
        firstName: true,
        lastName: true,
        themePref: true,
        createdAt: true,
      },
    });
    if (!user) throw new AppError('UNAUTHORIZED', 401);
    return user;
  }
}

import { CanActivate, ExecutionContext, Injectable } from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import type { Request } from 'express';
import { AppError } from '../common/errors/app-error';
import type { AuthUser } from './auth.types';

@Injectable()
export class JwtAuthGuard implements CanActivate {
  constructor(private readonly jwt: JwtService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const req = context.switchToHttp().getRequest<Request & { user?: AuthUser }>();
    const header = req.headers.authorization;

    if (!header?.startsWith('Bearer ')) {
      throw new AppError('UNAUTHORIZED', 401);
    }

    try {
      const payload = await this.jwt.verifyAsync<{
        sub: string;
        role: AuthUser['role'];
        isAdmin: boolean;
        jti?: string;
      }>(header.slice('Bearer '.length));

      req.user = {
        id: payload.sub,
        role: payload.role,
        isAdmin: payload.isAdmin,
        jti: payload.jti,
      };
      return true;
    } catch {
      throw new AppError('UNAUTHORIZED', 401);
    }
  }
}

import { Injectable } from '@nestjs/common';
import { Prisma, UserRole } from '@prisma/client';
import * as mammoth from 'mammoth';
import { PrismaService } from '../prisma/prisma.service';
import { AppError } from '../common/errors/app-error';
import type { AuthUser } from '../auth/auth.types';
import { CreateProfileDto } from './dto/create-profile.dto';
import { UpdateProfileDto } from './dto/update-profile.dto';
import { BASE_FILENAME_REGEX } from './profiles.constants';

const PROFILE_LIMITS: Record<UserRole, number | null> = {
  USER: 1,
  PRO: 3,
  AGENCY: null,
  ADMIN: null,
};

@Injectable()
export class ProfilesService {
  constructor(private readonly prisma: PrismaService) {}

  async create(user: AuthUser, dto: CreateProfileDto) {
    const limit = PROFILE_LIMITS[user.role];
    const count = await this.prisma.profile.count({
      where: { userId: user.id },
    });

    if (limit !== null && count >= limit) {
      throw new AppError(
        'PROFILE_LIMIT_REACHED',
        402,
        `Your plan allows ${limit} profile(s).`,
      );
    }

    const isDefault = dto.isDefault ?? count === 0;

    return this.prisma.$transaction(async (tx) => {
      if (isDefault) {
        await tx.profile.updateMany({
          where: { userId: user.id, isDefault: true },
          data: { isDefault: false },
        });
      }

      return tx.profile.create({
        data: {
          userId: user.id,
          name: dto.name,
          headerFullName: dto.headerFullName,
          headerCity: dto.headerCity,
          headerPhones: dto.headerPhones,
          headerEmail: dto.headerEmail,
          linkedIn: dto.linkedIn ?? null,
          isDefault,
        },
      });
    });
  }

  findAll(user: AuthUser) {
    return this.prisma.profile.findMany({
      where: { userId: user.id },
      orderBy: { id: 'asc' },
      include: { settings: true },
    });
  }

  async update(user: AuthUser, profileId: string, dto: UpdateProfileDto) {
    await this.getOwnedProfile(user, profileId);

    const data: Prisma.ProfileUpdateInput = {};
    if (dto.name !== undefined) data.name = dto.name;
    if (dto.headerFullName !== undefined) data.headerFullName = dto.headerFullName;
    if (dto.headerCity !== undefined) data.headerCity = dto.headerCity;
    if (dto.headerPhones !== undefined) data.headerPhones = dto.headerPhones;
    if (dto.headerEmail !== undefined) data.headerEmail = dto.headerEmail;
    if (dto.linkedIn !== undefined) data.linkedIn = dto.linkedIn;

    return this.prisma.$transaction(async (tx) => {
      if (dto.isDefault === true) {
        await tx.profile.updateMany({
          where: { userId: user.id, isDefault: true },
          data: { isDefault: false },
        });
        data.isDefault = true;
      } else if (dto.isDefault === false) {
        data.isDefault = false;
      }

      return tx.profile.update({ where: { id: profileId }, data });
    });
  }

  async uploadBase(
    user: AuthUser,
    profileId: string,
    file?: Express.Multer.File,
  ) {
    if (!file) throw new AppError('FILE_REQUIRED', 400, 'A .docx file is required.');
    await this.getOwnedProfile(user, profileId);

    if (!BASE_FILENAME_REGEX.test(file.originalname)) {
      throw new AppError(
        'INVALID_FILENAME',
        422,
        'Filename must match: First_Last_Role.docx (R18).',
      );
    }

    const { value } = await mammoth.extractRawText({ buffer: file.buffer });
    const text = value.trim();
    if (!text) throw new AppError('EMPTY_DOCUMENT', 422, 'The document contains no extractable text.');

    const last = await this.prisma.baseResume.findFirst({
      where: { profileId },
      orderBy: { version: 'desc' },
      select: { version: true },
    });
    const version = (last?.version ?? 0) + 1;

    return this.prisma.$transaction(async (tx) => {
      await tx.baseResume.updateMany({
        where: { profileId, isActive: true },
        data: { isActive: false },
      });

      const base = await tx.baseResume.create({
        data: {
          profileId,
          version,
          fileName: file.originalname,
          parsedJson: { text, chars: text.length },
          isActive: true,
        },
      });

      await tx.profile.update({
        where: { id: profileId },
        data: { activeBaseId: base.id },
      });

      return base;
    });
  }

  private async getOwnedProfile(user: AuthUser, profileId: string) {
    const profile = await this.prisma.profile.findFirst({
      where: { id: profileId, userId: user.id },
    });
    if (!profile) throw new AppError('PROFILE_NOT_FOUND', 404);
    return profile;
  }
}

import { Injectable } from '@nestjs/common';
import * as cheerio from 'cheerio';
import * as mammoth from 'mammoth';
import { PrismaService } from '../prisma/prisma.service';
import { AppError } from '../common/errors/app-error';
import type { AuthUser } from '../auth/auth.types';
import { CreateJdDto } from './dto/create-jd.dto';

const MIN_JD_CHARS = 200;

@Injectable()
export class JdsService {
  constructor(private readonly prisma: PrismaService) {}

  async create(
    user: AuthUser,
    dto: CreateJdDto,
    file?: Express.Multer.File,
  ) {
    const profile = await this.prisma.profile.findFirst({
      where: { id: dto.profileId, userId: user.id },
    });
    if (!profile) throw new AppError('PROFILE_NOT_FOUND', 404);

    const sourceType = file ? 'file' : dto.sourceType;
    if (!sourceType) {
      throw new AppError('SOURCE_TYPE_REQUIRED', 400, 'sourceType is required.');
    }

    let text: string;
    if (sourceType === 'paste') {
      if (!dto.text) throw new AppError('JD_TEXT_REQUIRED', 400);
      text = dto.text;
    } else if (sourceType === 'url') {
      if (!dto.url) throw new AppError('JD_URL_REQUIRED', 400);
      text = await this.fetchUrlText(dto.url);
    } else {
      if (!file) throw new AppError('JD_FILE_REQUIRED', 400);
      text = await this.fileText(file);
    }

    text = this.normalize(text);
    if (text.length < MIN_JD_CHARS) {
      throw new AppError(
        'JD_UNUSABLE',
        422,
        `Extracted job description must contain at least ${MIN_JD_CHARS} characters.`,
      );
    }

    const jd = await this.prisma.jobDescription.create({
      data: {
        userId: user.id,
        profileId: profile.id,
        sourceType,
        sourceUrl: dto.url ?? null,
        sourceFileName: file?.originalname ?? null,
        rawText: text,
        extractedChars: text.length,
        status: 'EXTRACTED',
      },
    });

    return {
      id: jd.id,
      status: jd.status,
      sourceType: jd.sourceType,
      extractedChars: jd.extractedChars,
      createdAt: jd.createdAt,
    };
  }

  async findOne(user: AuthUser, id: string) {
    const jd = await this.prisma.jobDescription.findFirst({
      where: { id, userId: user.id },
    });
    if (!jd) throw new AppError('JD_NOT_FOUND', 404);
    return jd;
  }

  private async fetchUrlText(rawUrl: string): Promise<string> {
    let url: URL;
    try {
      url = new URL(rawUrl);
    } catch {
      throw new AppError('INVALID_URL', 400, 'Invalid URL.');
    }
    if (!['http:', 'https:'].includes(url.protocol)) {
      throw new AppError('INVALID_URL', 400, 'Only http/https URLs are supported.');
    }

    const res = await fetch(url, {
      signal: AbortSignal.timeout(10_000),
      headers: { 'user-agent': 'ResumeEngineBot/1.0' },
    });
    if (!res.ok) {
      throw new AppError('URL_FETCH_FAILED', 422, `URL returned ${res.status}.`);
    }

    const html = await res.text();
    const $ = cheerio.load(html);
    $('script, style, noscript, svg').remove();
    return $('body').text();
  }

  private async fileText(file: Express.Multer.File): Promise<string> {
    const name = file.originalname.toLowerCase();
    if (name.endsWith('.docx')) {
      const { value } = await mammoth.extractRawText({ buffer: file.buffer });
      return value;
    }
    if (name.endsWith('.txt')) {
      return file.buffer.toString('utf8');
    }
    throw new AppError(
      'UNSUPPORTED_FILE',
      422,
      'Only .docx and .txt files are supported.',
    );
  }

  private normalize(input: string): string {
    return input
      .replace(/\r\n/g, '\n')
      .replace(/\u00a0/g, ' ')
      .replace(/[ \t]+/g, ' ')
      .replace(/\n{3,}/g, '\n\n')
      .trim();
  }
}

/**
 * Required by the Stage 1 schema contract:
 * Run.baseResumeId is required. When POST /runs is implemented (Stage 2),
 * call this helper so a missing baseResumeId defaults to the profile's
 * active BaseResume (isActive = true).
 */

import { PrismaService } from '../prisma/prisma.service';
import { AppError } from '../common/errors/app-error';

export async function resolveBaseResumeId(
  prisma: PrismaService,
  profileId: string,
  provided?: string | null,
): Promise<string> {
  if (provided) return provided;

  const active = await prisma.baseResume.findFirst({
    where: { profileId, isActive: true },
    orderBy: { version: 'desc' },
  });

  if (!active) {
    throw new AppError(
      'BASE_RESUME_REQUIRED',
      422,
      'No active base resume exists for this profile.',
    );
  }

  return active.id;
}
